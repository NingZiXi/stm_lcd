# Opt-in subdirectory: the application supplies its HAL interface target.
if(NOT TARGET stm32_hal)
    message(FATAL_ERROR "Provide an interface target stm32_hal with STM32F4 HAL include paths/defines")
endif()
option(STM_LCD_SPI_F4_DMA "Compile the optional normal-mode DMA transport" ON)
add_library(stm_lcd_spi_f4 STATIC stm_lcd_spi_f4.c)
target_include_directories(stm_lcd_spi_f4 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(stm_lcd_spi_f4 PUBLIC stm_lcd stm32_hal)
target_compile_features(stm_lcd_spi_f4 PUBLIC c_std_11)
if(STM_LCD_SPI_F4_DMA)
    target_sources(stm_lcd_spi_f4 PRIVATE stm_lcd_spi_f4_dma.c)
else()
    target_compile_definitions(stm_lcd_spi_f4 PRIVATE STM_LCD_SPI_F4_DMA=0)
endif()

add_library(stm_lcd_i2c_f4 STATIC stm_lcd_i2c_f4.c)
target_include_directories(stm_lcd_i2c_f4 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(stm_lcd_i2c_f4 PUBLIC stm_lcd stm32_hal)
target_compile_features(stm_lcd_i2c_f4 PUBLIC c_std_11)
