cmake_minimum_required(VERSION 3.22)
project(stm_lcd LANGUAGES C)
if(NOT TARGET stm_common)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../stm_common/CMakeLists.txt")
        add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../stm_common" "${CMAKE_CURRENT_BINARY_DIR}/stm_common")
    else()
        message(FATAL_ERROR "stm_lcd requires stm_common: provide its target or sibling source directory")
    endif()
endif()
add_library(stm_lcd STATIC src/stm_lcd_io.c src/stm_lcd_panel.c src/stm_lcd_touch.c)
target_include_directories(stm_lcd PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(stm_lcd PUBLIC stm_common)
target_compile_features(stm_lcd PUBLIC c_std_11)
